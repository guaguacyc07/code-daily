```java
class Trie {
    // 本质是一个多叉树
    // 将字符化为索引 a--0, b--1

    // 孩子节点数组: 共26个位置, 分别对应26个小写字母
    Trie[] son;  
    // 标记当前节点是否是一个单词的结尾
    boolean isEnd;

    public Trie() {
        son = new Trie[26];  // 一共26个字母
        isEnd = false;  // 默认不是末尾(当插入word时才会存在有末尾)
    }
    
    public void insert(String word) {
        // 获取调用这个方法的那个对象实例(本质上是根节点)
        Trie node = this;
        // 将word拆开为一个个的字符并遍历
        for (int i = 0; i < word.length(); i++) {
            // 获取当前遍历的字符
            char ch = word.charAt(i);
            // 转为索引
            int idx = ch - 'a';
            // 判断: 当前节点的下个字符是否存在, 不存在则添加, 存在则直接复用
            if (node.son[idx] == null) {
                node.son[idx] = new Trie();
            }
            // 移动当前节点
            node = node.son[idx];
        }
        // 遍历完后把当前节点设置为end (isEnd = true)
        node.isEnd = true;
    }
    
    public boolean search(String word) {
        // 查询的话, 只需要判断2件事情
        // 1. 是否有完整的子树链路
        // 2. 末尾节点是否isEnd == true 
        Trie node = this;
        for (int i = 0; i < word.length(); i++) {
            char ch = word.charAt(i);
            int idx = ch - 'a';
            // 判断是否有完整子树链路(没有就直接返回false, 有就不管)
            if(node.son[idx] == null) return false;
            // 移动当前节点
            node = node.son[idx];
        }
        // 判断是否isEnd == true (是就返回true, 不是就返回false)
        return node.isEnd;
    }
    
    public boolean startsWith(String prefix) {
        // 判断前缀就更简单了, 只需要管有无完整的子树链路即可
        Trie node = this;
        for (int i = 0; i < prefix.length(); i++) {
            char ch = prefix.charAt(i);
            int idx = ch - 'a';
            // 判断是否有完整子树链路(没有就直接返回false, 有就不管)
            if(node.son[idx] == null) return false;
            // 移动当前节点
            node = node.son[idx];
        }
        return true;
    }
}

/**
 * Your Trie object will be instantiated and called as such:
 * Trie obj = new Trie();
 * obj.insert(word);
 * boolean param_2 = obj.search(word);
 * boolean param_3 = obj.startsWith(prefix);
 */
```

- **时间复杂度：O(m)**

  m 为单词（或前缀）的长度。`insert`、`search`、`startsWith` 都是沿树往下走，最多走 m 层；每层取子节点是 `son[ch - 'a']` 的数组下标访问，O(1)。所以三个方法单次操作都是 O(m)。

  若一共插入 n 个单词、总字符数为 L，把整棵树建起来的总时间复杂度是 O(L)。

- **空间复杂度：O(N·26) = O(N)**

  N 为 Trie 的节点总数，最多是"所有被插入单词的总字符数 + 1"（加的是根节点）。每个节点固定开一个长度 26 的数组，即 26N 个槽位引用；26 是小写字母表大小，属常数，所以记作 O(N)。

  数组写法的代价就是这 26 倍常数：某个节点哪怕只有 1 个孩子，剩下 25 个槽位也照样占着。换成 `HashMap<Character, Trie>` 能省空间，但每次找孩子多一层哈希开销。
